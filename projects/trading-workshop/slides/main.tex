\section{Introduction to Trading}
% #####################################################################
% Short intro (a condensed version of Intro to Quant). Leave empty for now.

\begin{frame}{What is Trading?}
  \begin{sqt:definition}
    \textbf{Trading} is exchanging an asset for an agreed price. Every trade
    has two sides: a \textbf{buyer} and a \textbf{seller}.
  \end{sqt:definition}

  \begin{itemize}
    \item A trade happens only when someone willing to buy meets someone
      willing to sell --- they \emph{agree on a price}, not on value.
    \item The \textbf{price} is just the number they agree on right now; the
      \textbf{value} is what the asset is really worth.
    \item A trader's job: buy when the price is \emph{below} fair value, sell
      when it is \emph{above}. The gap is your \textbf{edge}.
  \end{itemize}

  \vspace{0.2cm}
  \textcolor{sqt-question}{Today: we estimate fair values and trade them live
  on the engine.}
\end{frame}
\note{
  Open here. Trading sounds mysterious but it is just buying and selling at a
  price two people agree on. Stress the difference between price and value:
  the whole game is having a better estimate of value than the person on the
  other side. Everything today comes back to this one idea --- find fair value,
  then buy cheap / sell dear.
}

\begin{frame}{Buying and Selling}
  \begin{itemize}
    \item \textbf{Bid} --- the highest price someone is willing to \textbf{buy} at.
    \item \textbf{Ask} (offer) --- the lowest price someone is willing to \textbf{sell} at.
    \item The gap between them is the \textbf{spread}.
    \item \textbf{Making a market} = quoting \emph{both} a bid and an ask, e.g.
      ``17 at 19'' --- I'll buy at 17, sell at 19.
  \end{itemize}

  \vspace{0.2cm}
  \begin{sqt:definition}
    You \textbf{buy} by hitting someone's ask; you \textbf{sell} by hitting
    their bid. A tighter spread means you are more confident in your value.
  \end{sqt:definition}
\end{frame}
\note{
  Give the concrete example: if I quote ``17 at 19'', I am telling the room I
  will buy at 17 and sell at 19. A market maker earns the spread; a taker pays
  it for immediacy. Tie the spread to confidence --- the surer you are of fair
  value, the tighter you can quote without getting picked off.
}

\begin{frame}{Position and Max Position}
  \begin{itemize}
    \item \textbf{Long} --- you have bought; you profit if the value goes \emph{up}.
    \item \textbf{Short} --- you have sold; you profit if the value goes \emph{down}.
    \item Your \textbf{position} is the net number of contracts you hold.
    \item $\text{PnL} = \text{position} \times (\text{final value} - \text{your entry price})$.
  \end{itemize}

  \vspace{0.2cm}
  \begin{sqt:definition}
    The engine enforces a \textbf{max position}: a cap on how long or short you
    can go. It bounds your risk --- you cannot bet the farm on one view.
  \end{sqt:definition}
\end{frame}
\note{
  Long = own it, short = sold it (you can sell something you don't have yet and
  buy it back later). Position is just your net exposure. Walk the PnL formula
  slowly with a number: bought 5 lots at 17, settles at 20, that's 5 times 3 =
  +15. The max position is the guardrail --- it keeps one bad view from blowing
  anyone up and forces people to actually price, not gamble.
}

\section{Trading Sim 1: Sum of Five Dice}
% #####################################################################

\begin{frame}{Trading Sim 1: Sum of Five Dice}
  \begin{sqt:definition}
    The true value of the asset is the \textbf{sum of five dice rolls}.
    $$V = d_1 + d_2 + d_3 + d_4 + d_5, \qquad d_i \sim \mathrm{Unif}\{1,\dots,6\}.$$
  \end{sqt:definition}

  \begin{itemize}
    \item Make a market on $V$ in the trading engine.
    \item Everyone can see the dice --- there is no hidden information yet.
    \item At the end, positions are liquidated at the true $V$.
  \end{itemize}

  \vspace{0.3cm}
  \textcolor{sqt-question}{What is a fair value for $V$? How wide should your market be?}
\end{frame}

\begin{frame}{The Five Dice}
  \begin{center}
    \vspace{0.4cm}
    % TODO: set the rolled values here (currently 3, 5, 2, 6, 4).
    % Dice are revealed one at a time; the sum appears on the last click.
    \only<1->{\die{3}}\hfill
    \only<-1>{\diehidden}\only<2->{\die{5}}\hfill
    \only<-2>{\diehidden}\only<3->{\die{2}}\hfill
    \only<-3>{\diehidden}\only<4->{\die{6}}\hfill
    \only<-4>{\diehidden}\only<5->{\die{4}}

    \vspace{0.8cm}
    \onslide<6->{{\Large $3 + 5 + 2 + 6 + 4 = \mathbf{20}$}}
  \end{center}
\end{frame}

\begin{frame}{Fair Value}
  \begin{sqt:definition}
    \textbf{Fair value} is the expected settlement value:
    $\mathbb{E}[V]$. It is the natural centre of your market.
  \end{sqt:definition}

  \begin{itemize}
    \item One die: $\mathbb{E}[d] = \tfrac{1+2+\dots+6}{6} = 3.5$.
    \item Five dice: $\mathbb{E}[V] = 5 \times 3.5 = \mathbf{17.5}$.
    \item Spread: how wide you quote reflects your \emph{uncertainty}. One die
      has $\mathrm{SD} \approx 1.7$; five dice $\mathrm{SD} \approx 3.8$ --- but
      the mean is known exactly, so you can quote tightly around $17.5$.
  \end{itemize}

  \vspace{0.2cm}
  \textcolor{sqt-question}{Quote around $17.5$. If someone sells you at $16$,
  take it --- you have $1.5$ of edge on average.}
\end{frame}
\note{
  This is the punchline of Sim 1. Fair value is just the expected value. Show
  the two-line calculation --- 3.5 per die, times five, is 17.5. Then the key
  intuition: because everyone can see the dice there is no hidden information,
  so the market should converge right onto 17.5 and spreads should be tight.
  Contrast this with the next game, where information is hidden and spreads
  should widen. Reward anyone who bought below or sold above 17.5.
}

\section{Trading Sim 2: Estimation with Reveals}
% #####################################################################
% An estimation market where information is revealed over rounds.

\begin{frame}{Trading Sim 2: Estimation Game}
  \begin{sqt:definition}
    The true value is some real-world quantity that nobody knows exactly.
    Clues are \textbf{revealed round by round}; update your market as you learn more.
  \end{sqt:definition}

  \vspace{0.3cm}
  \textcolor{sqt-question}{Open a market on each. Positions liquidate at the true value.}
\end{frame}
\note{
  Now the fun part --- nobody knows these values exactly, so it is pure
  estimation. The skill is bounding: you may not know the answer, but you know
  it is not 10 and not a million. Reveal clues one at a time and watch the
  market move toward the truth. Encourage people to trade on every new clue,
  not just wait for certainty --- that is how real markets absorb news.
}

\begin{frame}{Estimation Game: An Integral}
  \begin{sqt:definition}
    The asset is the value of
    $$V = \int_{1}^{100} \ln(x)\,\mathrm{d}x.$$
  \end{sqt:definition}

  \onslide<2->{\textbf{Hint}}
  \begin{itemize}
    \item<2-> $\ln(x)$ grows slowly: $\ln(1) = 0$, $\ln(10) \approx 2.3$, $\ln(100) \approx 4.6$.
    \item<3-> An antiderivative is $\int \ln(x)\,\mathrm{d}x = x\ln(x) - x$.
  \end{itemize}

  \onslide<4->{
    \vspace{0.2cm}
    \begin{sqt:result}
      \textbf{True value:}
      $\big[x\ln(x) - x\big]_{1}^{100} = 100\ln(100) - 99 \approx \mathbf{361.5}$.
    \end{sqt:result}
  }
\end{frame}
\note{
  Even if they have never seen the antiderivative, they can bound it: the
  average height of $\ln$ between 1 and 100 is somewhere around 3.5--4, times a
  width of 99, so a few hundred. That estimate alone gets them close to 361.5.
  The lesson: you rarely need the exact formula to make a good market --- a
  rough bound plus an average is enough to quote a sensible price.
}

\section{Options}
% #####################################################################

\begin{frame}{What is an Option?}
  \begin{sqt:definition}
    An \textbf{option} is a contract whose value depends on some underlying
    asset, but where one side has a \textbf{choice} at expiry.
  \end{sqt:definition}

  \begin{itemize}
    \item ``Right, not obligation'' is the key idea:
      if exercise is unfavourable for you, you simply do nothing.
    \item The payoff is some function of the final underlying value:
      $g(S_T)$.
    \item Fair price means averaging that payoff across all possible outcomes.
  \end{itemize}

  \vspace{0.2cm}
  \textcolor{sqt-question}{An option gives you a choice. How much is that
  choice worth?}
\end{frame}
\note{
  Start from the literal meaning of the word option. It is a choice. You are
  not forced to trade at expiry; you only exercise when it helps you. That
  asymmetry is the entire product. Keep this slide generic: payoff is just some
  function of the final price, and pricing means averaging that function over
  the distribution of outcomes.
}

\begin{frame}{Pricing an Option from the Underlying Distribution}
  \vspace{0.2cm}
  \begin{sqt:definition}
    \[
      \text{Option price}
      = e^{-rT}\,\mathbb{E}\big[g(S_T)\big]
      = e^{-rT}\int_{-\infty}^{\infty} g(s)\, f(s)\, ds
    \]
  \end{sqt:definition}

  \begin{itemize}
    \item $S_T$: underlying price at expiry time $T$.
    \item $g(S_T)$: option payoff as a function of the expiry price.
    \item $f(s)$: probability density of the underlying at expiry.
    \item $r$: interest rate used for discounting.
    \item $T$: time to expiry.
  \end{itemize}
\end{frame}
\note{
  Keep this slide formal and compact. Define each term clearly, and mention
  that the key structure is discounted expected payoff under the expiry
  distribution.
}

\begin{frame}{Calls and Puts: Definition and Payoff}
  At expiry, the two basic option payoffs are:

  \vspace{0.2cm}
  \begin{columns}[t]
    \begin{column}{0.5\textwidth}
      \centering
      \payoff{
        \draw[very thick,sqt-question] (0,0) -- (3,0) -- (6,3);
        \draw[dashed,gray!50] (3,0) -- (3,-0.2);
        \node[below] at (3,-0.2) {\scriptsize $K$};
      }

      \textbf{Long call} $= \max(S-K,0)$

      \vspace{0.15cm}
      Exercise if $S>K$.
    \end{column}
    \begin{column}{0.5\textwidth}
      \centering
      \payoff{
        \draw[very thick,orange!85!black] (0,3) -- (3,0) -- (6,0);
        \draw[dashed,gray!50] (3,0) -- (3,-0.2);
        \node[below] at (3,-0.2) {\scriptsize $K$};
      }

      \textbf{Long put} $= \max(K-S,0)$

      \vspace{0.15cm}
      Exercise if $S<K$.
    \end{column}
  \end{columns}

  \vspace{0.3cm}
  \begin{itemize}
    \item A long call benefits when the underlying settles at a
      \textbf{higher} price.
    \item A long put benefits when the underlying settles at a
      \textbf{lower} price.
    \item Selling the option just flips the payoff sign.
  \end{itemize}
\end{frame}
\note{
  This is the one slide where they should leave knowing the two canonical
  pictures. Call: flat then up. Put: down then flat. Also say out loud that
  the payoff diagram is at expiry and does not include the initial premium.
}

\begin{frame}{Call Option Pricing Example}
  \begin{sqt:definition}
    A single die is rolled, value $d \sim \mathrm{Unif}\{1,\dots,6\}$.
    The asset is a \textbf{call option} with strike $K = 3$:
    $$\text{payoff} = \max(d - 3,\; 0).$$
  \end{sqt:definition}

  \begin{center}
    \diehidden

    \vspace{0.4cm}
    \begin{tabular}{c|cccccc}
      $d$ & 1 & 2 & 3 & 4 & 5 & 6 \\ \hline
      $\max(d-3,0)$ & 0 & 0 & 0 & 1 & 2 & 3 \\
    \end{tabular}
  \end{center}

  \vspace{0.3cm}
  \textcolor{sqt-question}{What is the fair price of this option?}
  % Fair value = (0+0+0+1+2+3)/6 = 1.
\end{frame}
\note{
  First option they price by hand. Walk the payoff row: 0,0,0,1,2,3. Average it
  --- that's 6 over 6 = 1. So the fair price is exactly 1. Notice it is worth
  something even though half the time it pays zero: the upside outcomes carry
  all the value. That is the whole idea of an option in one number.
}

\section{Trading Sim 3}

\begin{frame}{Trading Sim 3}
  \begin{sqt:definition}
    Three independent random integers are generated:
    \[
      X_1,X_2,X_3 \sim \operatorname{Unif}\{0,1,\ldots,1000\}.
    \]

    The underlying is the minimum:
    \[
      S=\min(X_1,X_2,X_3).
    \]
  \end{sqt:definition}

  \begin{itemize}
    \item The traded contract is a \textbf{put option} with strike $K=200$.
    \item Final settlement value:
      \[
        (200-S)^+ = \max(200-S,0).
      \]
    \item The put pays when at least one generated number is sufficiently low.
  \end{itemize}

  \vspace{0.2cm}
  \textcolor{sqt-question}{Make an initial market before any numbers are revealed.}
\end{frame}
\note{
  The key feature is that the underlying is the worst of three draws, not the
  average or the sum. That makes downside outcomes much more relevant. Let them
  feel that intuition before you say anything quantitative.
}

\begin{frame}{Trading Sim 3}
  \begin{center}
    \[
      X_1=734,\qquad X_2=?,\qquad X_3=?
    \]
  \end{center}

  \begin{itemize}
    \item One number is now known.
    \item The other two numbers still determine whether the minimum falls below $200$.
    \item Re-price the \textbf{200-strike put}.
  \end{itemize}
\end{frame}
\note{
  A high first number does not help much because the minimum can still be driven
  by either hidden draw. So the downside risk is still very alive.
}

\begin{frame}{Trading Sim 3}
  \begin{center}
    \[
      X_1=734,\qquad X_2=168,\qquad X_3=?
    \]

    \vspace{0.2cm}
    Current minimum:
    \[
      \min(734,168)=168
    \]

    \vspace{0.2cm}
    Guaranteed payoff so far:
    \[
      200-168=32
    \]
  \end{center}
\end{frame}
\note{
  Once a number below strike appears, the option has locked in intrinsic value.
  The final draw cannot hurt the holder anymore; it can only keep the payoff at
  32 or make it larger by setting a new, even lower minimum.
}

\begin{frame}{Trading Sim 3}
  \begin{center}
    \[
      X_1=734,\qquad X_2=168,\qquad X_3=491
    \]

    \vspace{0.2cm}
    \[
      S=\min(734,168,491)=168
    \]

    \vspace{0.3cm}
    {\Large
    \[
      \text{Put payoff}=\max(200-168,0)=32
    \]}
  \end{center}
\end{frame}
\note{
  Settle it cleanly. The third draw was not low enough to change the minimum,
  so the guaranteed payoff from the previous slide becomes the final payoff.
}

\begin{frame}{Trading Sim 3}
  \begin{sqt:definition}
    Continuous approximation:
    \[
      \mathbb P(S>x)=\left(\frac{1000-x}{1000}\right)^3
    \]
  \end{sqt:definition}

  \[
    \mathbb E[(200-S)^+]
    =
    \int_0^{200}
    \left[
      1-\left(1-\frac{x}{1000}\right)^3
    \right]dx
    = 52.8
  \]

  \vspace{0.2cm}
  \textcolor{sqt-question}{Initial theoretical fair value: approximately 53.}
\end{frame}
\note{
  This is where you can give the fair value. A clean continuous approximation
  is to treat the draws as i.i.d. uniform on [0,1000]. Then
  $\mathbb{P}(S>x)=((1000-x)/1000)^3$, and integrating the put payoff over
  $x\in[0,200]$ gives about 52.8. The important intuition is that taking the
  minimum of three draws pushes probability mass toward low outcomes, which is
  exactly what makes the put valuable.
}

\section{Trading Sim 4}

\begin{frame}{Trading Sim 4}
  \begin{sqt:definition}
    A fair coin is flipped \textbf{before} the cards are drawn. Then two cards
    $X$ and $Y$ are drawn \textbf{without replacement} from
    $\{1,2,\dots,14\}$.
  \end{sqt:definition}

  \[
    S=
    \begin{cases}
      XY, & \text{if Heads},\\
      X+Y, & \text{if Tails.}
    \end{cases}
  \]

  \begin{itemize}
    \item The underlying value is $S$.
    \item The traded contract is a \textbf{25-strike straddle}.
    \item Final settlement value: $\lvert S-25\rvert$.
  \end{itemize}

  \vspace{0.2cm}
  \textcolor{sqt-question}{Make an initial market before any information is revealed.}
\end{frame}
\note{
  Set up the game cleanly. There are two sources of uncertainty: the rule and
  the cards. Stress that the coin is flipped first but hidden, so the room has
  to price over a mixture distribution from the start.
}

\begin{frame}{Trading Sim 4}
  \begin{center}
    \[
      X=8,\qquad Y=?,\qquad \text{rule}=?
    \]
  \end{center}

  \begin{itemize}
    \item One card is now known.
    \item The rule and the second card are still hidden.
    \item Re-price the \textbf{25-strike straddle}.
  \end{itemize}

  \vspace{0.2cm}
  \textcolor{sqt-question}{How does learning $X=8$ change your market?}
\end{frame}
\note{
  They still face a mixture of addition and multiplication, but the first card
  starts to anchor the range of outcomes. Let them talk through what values are
  now possible under each rule.
}

\begin{frame}{Trading Sim 4}
  \begin{center}
    \[
      X=8,\qquad Y=?,\qquad \text{Heads: } S=XY
    \]
  \end{center}

  \begin{itemize}
    \item Multiplication can create much \textbf{larger} outcomes.
    \item It also creates a more \textbf{dispersed} distribution of $S$.
    \item Addition would have produced a much \textbf{narrower} range.
  \end{itemize}

  \vspace{0.2cm}
  \textcolor{sqt-question}{Update your market again with the rule now known.}
\end{frame}
\note{
  This is the key information shock. Same first card, but knowing the rule
  changes the whole shape of the distribution. That is exactly what the straddle
  should respond to.
}

\begin{frame}{Trading Sim 4}
  \begin{center}
    \[
      X=8,\qquad Y=12,\qquad \text{Heads: } S=XY
    \]

    \vspace{0.2cm}
    \[
      S = 8\times 12 = 96
    \]

    \vspace{0.2cm}
    \[
      \text{Payoff} = |96-25| = 71
    \]
  \end{center}
\end{frame}
\note{
  Reveal the second card, compute the underlying, then settle the straddle.
  This example lands hard because multiplication pushes the final value far from
  the strike.
}

\begin{frame}{Trading Sim 4}
  \begin{sqt:definition}
    Exact fair value:
    \[
      \mathbb{E}[|S-25|]
      =
      \frac{1}{2\binom{14}{2}}
      \sum_{1\le x<y\le 14}
      \Bigl(|xy-25| + |x+y-25|\Bigr)
      \approx 24.01
    \]
  \end{sqt:definition}

  \begin{itemize}
    \item Choose the coin outcome: Heads or Tails, each with probability $\frac12$.
    \item Choose the two cards uniformly without replacement from $\{1,\dots,14\}$.
    \item Compute the straddle payoff for each case and average them all.
  \end{itemize}
\end{frame}
\note{
  Keep this one discrete and exact. There are $\binom{14}{2}$ unordered card
  pairs, and for each pair there are two equally likely rules. So the true fair
  value is just the average payoff across all those equally likely outcomes.
}

\section{Wrap Up}
% #####################################################################

\begin{frame}{Wrap Up}
  \textbf{What we did today:}
  \begin{itemize}
    \item \textbf{Fair value} is expected value --- centre your market on it.
    \item \textbf{Spreads} price your uncertainty; hidden information widens them.
    \item \textbf{Estimation} under uncertainty --- update your market as clues land.
    \item \textbf{Options} price one-sided payoffs, $\mathbb{E}[\max(\cdot,0)]$,
      and let you trade the \textbf{shape of the distribution}.
  \end{itemize}

  \vspace{0.2cm}
  \begin{sqt:definition}
    \textbf{Takeaways for interviews:} think in expected value, quote a number
    and a width, and re-price the instant new information arrives.
  \end{sqt:definition}

  \vspace{0.2cm}
  \begin{center}
    \textcolor{sqt-question}{\large Thanks for trading with us! Questions?}
  \end{center}
\end{frame}
\note{
  Land the plane by tying the games together: every single asset today was
  priced the same way --- expected value --- whether it was dice, followers, an
  integral, or an option. Spreads were about how confident you were. Options
  just added the max. For interviews: they will ask you to make a market on
  something silly (golf balls in a bus); the move is always the same --- state
  a fair value, quote a width around it, and update fast. Thank them and open
  the floor.
}
